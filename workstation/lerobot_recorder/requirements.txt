# LeRobot recorder/replay — workstation env (SEPARATE from the YAM/robot env).
#
# The robot link is portal (plain TCP) via i2rt's RobotClient, so this
# env just needs i2rt installed (for i2rt.serving.robot_client) plus the deps
# below. The policy bridge additionally uses yam-policy (websocket client).
#
# System packages (apt) needed for LeRobot v3.0 video encoding:
#   sudo apt install -y ffmpeg
#
# Setup (conda env + uv installs; full guide in README.md):
#   conda create -y -n yam_ws python=3.11   # any Python >= 3.10
#   conda activate yam_ws
#   uv pip install -e .                      # i2rt (robot client, portal) — targets the conda env
#   uv pip install -e policy_serving         # yam-policy (websocket client, for the bridge)
#   uv pip install -r workstation/lerobot_recorder/requirements.txt
#   # another policy repo in the SAME env:  pip install -e /path/to/policy_repo
#
# RealSense udev rules are NOT installed by the pip wheel — see README.md
# (workstation -> RealSense cameras) or cameras fail with permission errors.

# Pinned, not floating: the recorder reaches into LeRobot internals that are not covered
# by its public API — episode video windows in meta/episodes, save_episode(parallel_encoding=),
# and _batch_save_episode_video for the trailing-batch flush. A minor bump can move those
# silently. Re-pin deliberately after checking workstation/lerobot_recorder against the
# new release; the video-integrity checks in dataset_writer will flag encode regressions.
lerobot==0.4.4      # LeRobotDataset v3.0 API (create/add_frame/save_episode/finalize)
torchcodec>=0.9     # VideoEncoder API used by the workstation recording backend
pyrealsense2        # RealSense D405 / D455 cameras
PyQt5               # GUI
numpy

# `yam-data render-samples` (workstation.lerobot_recorder.render_deploy_samples): writes the
# candidate-fan mp4. mujoco/mink for the FK it also needs come from the root `i2rt` package (-e
# .), not from here -- see pyproject.toml's unconditional dependencies.
imageio[ffmpeg]

# `yam-data calibrate-agentview` (workstation.lerobot_recorder.charuco): ChArUco board detection.
# >=4.7 specifically -- charuco.py uses the CharucoDetector/CharucoBoard.matchImagePoints API
# introduced then; it does not fall back to the pre-4.7 detectMarkers+interpolateCornersCharuco
# calls, which OpenCV removed. contrib, not the plain `opencv-python`, for cv2.aruco itself.
opencv-contrib-python>=4.7
